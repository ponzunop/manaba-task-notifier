コピー用 GAS コードを表示（複数人共有・公開HTTPS URL・gasState 保護対応） 
// 許可されたアプリ本体オリジンのみコールバックを許可（データ流出防止）
var ALLOWED_ORIGINS = [
  'http://localhost:8080',
  'http://127.0.0.1:8080',
  '__APP_ORIGIN__'
];

function collectManabaMessages(paramEmail, paramGasUrl) {
  var query = '(from:do-not-reply@manaba.jp OR from:manaba OR manaba) -コースニュース掲示のお知らせ -コースニュース';
  var threads = GmailApp.search(query, 0, 50);
  var emails = [];
  for (var i = 0; i < threads.length; i++) {
    var msgs = threads[i].getMessages();
    for (var j = 0; j < msgs.length; j++) {
      var m = msgs[j];
      emails.push({
        subject: m.getSubject(),
        from: m.getFrom(),
        date: m.getDate().toISOString(),
        body: m.getPlainBody()
      });
    }
  }
  return {
    email: Session.getActiveUser().getEmail() || paramEmail || '',
    gasUrl: ScriptApp.getService().getUrl() || paramGasUrl || '',
    emails: emails
  };
}

function escapeHtmlAttr(str) {
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

function resolveSafeReturnUrl(rawReturn) {
  var fallback = '__APP_ORIGIN__/auth/gas-callback';
  if (!rawReturn) {
    return fallback;
  }
  for (var i = 0; i < ALLOWED_ORIGINS.length; i++) {
    var origin = ALLOWED_ORIGINS[i];
    if (origin && rawReturn === origin + '/auth/gas-callback') {
      return rawReturn;
    }
  }
  return fallback;
}

function doGet(e) {
  var rawReturn = (e && e.parameter && e.parameter.returnUrl) ? e.parameter.returnUrl : '';
  var returnUrl = resolveSafeReturnUrl(rawReturn);
  var paramEmail = (e && e.parameter && e.parameter.email) ? e.parameter.email : '';
  var paramGasUrl = (e && e.parameter && e.parameter.gasUrl) ? e.parameter.gasUrl : '';
  var paramGasState = (e && e.parameter && e.parameter.gasState) ? e.parameter.gasState : '';

  var payload = collectManabaMessages(paramEmail, paramGasUrl);
  var payloadJson = JSON.stringify(payload);
  var count = payload.emails.length;
  var activeEmail = payload.email || '確認済みGoogleアカウント';

  var html = '<!DOCTYPE html><html lang="ja"><head><meta charset="UTF-8">'
    + '<meta name="viewport" content="width=device-width,initial-scale=1.0">'
    + '<style>body{font-family:sans-serif;max-width:540px;margin:40px auto;padding:20px;text-align:center;color:#0f172a;}'
    + '.btn{background:#2563eb;color:#fff;border:none;padding:14px 22px;border-radius:8px;font-size:16px;font-weight:bold;cursor:pointer;margin:8px;width:100%;max-width:420px;}'
    + '.btn2{background:#e2e8f0;color:#0f172a;}'
    + '.meta{color:#475569;font-size:14px;margin-bottom:16px;}</style></head><body>'
    + '<h2>manaba 通知メールを取得しました（' + count + '件）</h2>'
    + '<p class="meta">ログインアカウント: <strong>' + escapeHtmlAttr(activeEmail) + '</strong><br>下の青いボタンを押してアプリへ送信・ログインしてください。</p>'
    + '<form id="syncForm" method="POST" action="' + escapeHtmlAttr(returnUrl) + '" target="_top">'
    + '<input type="hidden" name="payload" value="' + escapeHtmlAttr(payloadJson) + '">'
    + (paramGasState ? '<input type="hidden" name="gasState" value="' + escapeHtmlAttr(paramGasState) + '">' : '')
    + '<button type="submit" class="btn">アプリへ送信してログインする（取得件数: ' + count + '件）</button>'
    + '</form>'
    + '<button type="button" class="btn btn2" onclick="copyJson()">JSONをコピー</button>'
    + '<script>'
    + 'var rawJson = ' + JSON.stringify(payloadJson) + ';'
    + 'function copyJson(){ navigator.clipboard.writeText(rawJson).then(function(){ alert("JSONをコピーしました"); }); }'
    + '</script></body></html>';

  return HtmlService.createHtmlOutput(html)
    .setTitle('manaba 課題通知システム 連携')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function exportManabaJson() {
  Logger.log(JSON.stringify(collectManabaMessages('', '')));
}

方法②：Google OAuth 2.0 + Gmail API で連携（設計書準拠・個人アカウント向け）

個人の Google アカウント等で Google Cloud Console を利用できる場合は、OAuth 2.0（gmail.readonly）で直接連携できます。
※ Google Cloud の OAuth 同意画面は HTTPS が必要ですが、OAuth クライアント ID の「承認済みのリダイレクト URI」には http://localhost:8080/auth/callback をそのまま登録できます。

ブラウザ経由 GAS 連携の仕組み

組織制限クリア
1

ブラウザが Google 認証済みセッションを活用

サーバから GAS を叩くと「全員」公開が必要ですが、本方式はログイン済みブラウザが GAS（「自分のみ」公開）を開いてフォーム POST するため安全です。

2

公開時間・締切時間を柔軟に自動抽出

[受付開始日時]・[受付終了日時]・■コース名・期間表記・曜日付き日時など多様な manaba 通知形式を解析します。

3

解析できないメールもその場で確認可能

manaba 関連メールが見つかり形式が特殊でスキップされた場合も、課題一覧画面のデバッグパネルで件名・本文を確認できます。

4

通知設定・自動リマインド・受験済管理

24時間前・5時間前・1時間前などの複数通知や科目別 ON/OFF、受験済/提出済ボタンによる通知停止まで一気通貫で動作します。
